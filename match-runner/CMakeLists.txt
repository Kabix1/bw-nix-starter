cmake_minimum_required(VERSION 3.20)
project(BWMatch LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(STARDUST "${BW_ENV_ROOT}/StardustDevEnvironment")
set(BWAPI_SOURCE "${STARDUST}/3rdparty/openbw/bwapi/bwapi")
set(BW_ENV_BUILD "${BW_ENV_ROOT}/build")
foreach(lib BWAPI BWAPILIB OpenBWData)
  find_library(${lib}_LIB NAMES ${lib} PATHS "${BW_ENV_BUILD}/lib" NO_DEFAULT_PATH REQUIRED)
endforeach()
find_library(DEMO_LIB NAMES DemoAIModule PATHS "${BW_ENV_BUILD}/src" NO_DEFAULT_PATH REQUIRED)
foreach(lib gtest gtest_main)
  find_library(${lib}_LIB NAMES ${lib} PATHS "${BW_ENV_BUILD}/test/3rdparty/googletest" NO_DEFAULT_PATH REQUIRED)
endforeach()
add_executable(bw-match Match.cpp BWTest.cpp "${STARDUST}/test/Maps.cpp")
target_include_directories(bw-match SYSTEM PRIVATE "${STARDUST}/test" "${STARDUST}/src" "${STARDUST}/src/Instrumentation" "${STARDUST}/test/3rdparty/googletest/include" "${BWAPI_SOURCE}/include" "${BWAPI_SOURCE}/Util/Source" "${BWAPI_SOURCE}/BWAPI/openbw" "${BWAPI_SOURCE}/BWAPI/openbw/BWAPI" "${BWAPI_SOURCE}/BWAPI/Source" "${BWAPI_SOURCE}/BWAPICore" "${BWAPI_SOURCE}/OpenBWData" "${BWAPI_SOURCE}/Shared" "${STARDUST}/3rdparty/openbw/openbw")
target_link_libraries(bw-match PRIVATE "${gtest_main_LIB}" "${gtest_LIB}" "${BWAPI_LIB}" "${BWAPILIB_LIB}" "${OpenBWData_LIB}" "${DEMO_LIB}")
